A throttled binding looks like this in React:

```tsx
import { useSearchParams } from 'kvant/react'
import * as kv from 'kvant/schema'
import { useThrottleFn } from 'react-use'

export function Search() {
  const [q, setQ] = useSearchParams('q', kv.string().default(''))
  const throttledSetQ = useThrottleFn(setQ, 300)

  return <input defaultValue={q} onChange={e => throttledSetQ(e.target.value)} />
}
```
