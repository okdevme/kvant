## Usage

```tsx
import { useCookies } from 'kvant/react'
import * as kv from 'kvant/schema'

export function ConsentBanner() {
  const [consent, setConsent] = useCookies(
    'consent',
    kv.enum(['accepted', 'declined']),
    { maxAge: 60 * 60 * 24 * 365 }, // one year
  )

  if (consent) return null

  return <button onClick={() => setConsent('accepted')}>Accept</button>
}
```

## SSR setup

In React Router [framework mode](https://reactrouter.com/start/framework/installation),
`document.cookie` is unavailable during server rendering. Pass the request's
`Cookie` header as the `fallback` option via `CookiesOptionsProvider`:

```tsx title="app/root.tsx"
import { CookiesOptionsProvider } from 'kvant/react'

export async function loader({ request }: { request: Request }) {
  return { cookies: request.headers.get('Cookie') ?? '' }
}

export default function Root({ loaderData, children }) {
  return (
    <CookiesOptionsProvider
      defaultOptions={{ fallback: loaderData.cookies }}
    >
      {children}
    </CookiesOptionsProvider>
  )
}
```
