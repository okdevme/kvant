<switch>
  <case react vue>
`useCookies` binds state to a [cookie](https://developer.mozilla.org/en-US/docs/Web/HTTP/Guides/Cookies). Cookies **can be read by server**,
which makes them ideal for values that affect SSR: locale, theme, feature flags.
Changes are picked up across tabs via the [Cookie Store API](https://developer.mozilla.org/en-US/docs/Web/API/Cookie_Store_API)
where available.
  </case>
  <case nuxt>
`useCookies` binds state to a [cookie](https://developer.mozilla.org/en-US/docs/Web/HTTP/Guides/Cookies), built on Nuxt's SSR-aware
[`useCookie`](https://nuxt.com/docs/api/composables/use-cookie). Cookies
work on both server and client with no fallback setup: the server reads the
request's [`Cookie` header](https://developer.mozilla.org/en-US/docs/Web/HTTP/Reference/Headers/Cookie) and the value is serialized into the payload for
hydration.
  </case>
</switch>

<switch>
  <case react>
```tsx twoslash
import { useCookies } from 'kvantjs/react'
import * as kv from 'kvantjs/schema'

export function LocaleSelect() {
  const [locale, setLocale] = useCookies('locale', kv.string().default('en'))

  return (
    <select value={locale} onChange={e => setLocale(e.target.value)}>
      <option value="en">English</option>
      <option value="de">Deutsch</option>
    </select>
  )
}
```
  </case>
  <case vue>
```vue twoslash
<script setup lang="ts">
import { useCookies } from 'kvantjs/vue'
import * as kv from 'kvantjs/schema'

const locale = useCookies('locale', kv.string().default('en'))
</script>

<template>
  <select v-model="locale">
    <option value="en">English</option>
    <option value="de">Deutsch</option>
  </select>
</template>
```
  </case>
  <case nuxt>
```vue twoslash
<script setup lang="ts">
import { useCookies } from 'kvantjs/nuxt'
import * as kv from 'kvantjs/schema'

const locale = useCookies('locale', kv.string().default('en'))
</script>

<template>
  <select v-model="locale">
    <option value="en">English</option>
    <option value="de">Deutsch</option>
  </select>
</template>
```
  </case>
</switch>

Key maps work too:

<switch>
  <case react>
```ts twoslash
import { useCookies } from 'kvantjs/react'
import * as kv from 'kvantjs/schema'
// ---cut---
const [prefs, setPrefs] = useCookies({
  locale: kv.string().default('en'),
  theme: kv.enum(['light', 'dark']).default('light'),
})

setPrefs(prev => ({ ...prev, theme: 'dark' }))
```
  </case>
  <case vue>
```ts twoslash
import { useCookies } from 'kvantjs/vue'
import * as kv from 'kvantjs/schema'
// ---cut---
const prefs = useCookies({
  locale: kv.string().default('en'),
  theme: kv.enum(['light', 'dark']).default('light'),
})

prefs.value = { ...prefs.value, theme: 'dark' }
prefs.value.theme = 'dark' // deeply reactive!
```
  </case>
</switch>
