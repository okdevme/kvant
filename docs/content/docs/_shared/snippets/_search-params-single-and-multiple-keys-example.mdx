title: Search Params Single and Multiple Keys Example

<switch>
  <case react>
```ts twoslash
import { useSearchParams } from 'kvantjs/react'
import * as kv from 'kvantjs/schema'
// ---cut---
const [page, setPage] = useSearchParams('page', kv.index().default(0))

const [filters, setFilters] = useSearchParams({
  q: kv.string().default(''),
  tags: kv.array(kv.string()).default([]),
  sort: kv.enum(['asc', 'desc']).default('asc'),
})
```
  </case>
