<switch>

```tsx twoslash
import { useSearchParams } from 'kvant/react'
import * as kv from 'kvant/schema'

function SearchInput() {
  const [query, setQuery] = useSearchParams('q', kv.string().default(''))

  return (
    <input
      value={query}
      onChange={e => setQuery(e.target.value)}
    />
  )
}
```
  </case>
