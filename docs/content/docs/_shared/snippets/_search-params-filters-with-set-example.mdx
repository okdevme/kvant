---
title: Search Params Filters Example
---

<switch>
  <case react>
```tsx
const [filters, setFilters] = useSearchParams({
  q: kv.string().default(''),
  page: kv.index().default(0),
  sort: kv.enum(['asc', 'desc']).default('asc'),
  tags: kv.string().array().default([]), // repeated params: ?tags=a&tags=b
})

setFilters(prev => ({ ...prev, page: prev.page + 1 }))
```
  </case>
  <case next-pages>
```tsx
const [filters, setFilters] = useRouterQuery({
  q: kv.string().default(''),
  page: kv.index().default(0),
  sort: kv.enum(['asc', 'desc']).default('asc'),
  tags: kv.string().array().default([]), // repeated params: ?tags=a&tags=b
})

setFilters(prev => ({ ...prev, page: prev.page + 1 }))
```
  </case>
  <case vue>
```tsx
const filters = useSearchParams({
  q: kv.string().default(''),
  page: kv.index().default(0),
  sort: kv.enum(['asc', 'desc']).default('asc'),
  tags: kv.string().array().default([]), // repeated params: ?tags=a&tags=b
})

filters.value = { ...filters.value, page: filters.value.page + 1 }
filters.value.page += 1 // deeply reactive!
```
  </case>
  <case vue-router nuxt>
```tsx
const filters = useRouteQuery({
  q: kv.string().default(''),
  page: kv.index().default(0),
  sort: kv.enum(['asc', 'desc']).default('asc'),
  tags: kv.string().array().default([]), // repeated params: ?tags=a&tags=b
})

filters.value = { ...filters.value, page: filters.value.page + 1 }
filters.value.page += 1 // deeply reactive!
```
  </case>
</switch>
